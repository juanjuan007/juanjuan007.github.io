[TIOJ 1868](https://tioj.ck.tp.edu.tw/problems/1224)

<br>

**題敘**

給 $N$ 個平面上的矩形，請求出它們覆蓋的總表面積。

$1 \leq N \leq 2\times 10^{5}$

<br>

**思路**

這題就是掃描線，掃過 $x$ 軸，不斷維護當前的狀況。

所以需要一個支援區間加值、區間查詢正數個數的線段樹。

由於區間內的數字不可能為負數，我們可以直接在每個節點紀錄最小值是多少及個數，用區間長度扣除是 $0$ 的個數就可以了，但還要支援區間加值，所以要打懶惰標記。

Time : $O(N \log N)$

Code:

```cpp
#include<bits/stdc++.h>
using namespace std;

const int SZ = 1e6+5;

struct Node{
    int mini,cnt,tag;
};

vector<Node>seg(4*SZ);

int get(int id){
    return(seg[id].mini + seg[id].tag);
}


void pull(int id){
    int  l = get(id*2+1),r = get(id*2+2);
    if(l<r){
        seg[id].mini = l;
        seg[id].cnt = seg[id*2+1].cnt;
    }else if(l==r){
        seg[id].mini = l;
        seg[id].cnt = seg[id*2+1].cnt + seg[id*2+2].cnt;
    }else{
        seg[id].mini = r;
        seg[id].cnt = seg[id*2+2].cnt;
    }
    seg[id].tag  = 0;
}

void push(int id){
    seg[id*2+1].tag += seg[id].tag;
    seg[id*2+2].tag += seg[id].tag;
    seg[id].mini = get(id);
    seg[id].tag = 0;
}

void Build(int l,int r,int id){
    if(l==r){
        seg[id] = {0,1,0};
        return;
    }
    int m = (l+r)/2;
    Build(l,m,id*2+1);
    Build(m+1,r,id*2+2);
    pull(id);
}

void update(int ql,int qr,int v,int l,int r,int id){
    if(r<ql || qr<l) return;
    if(ql<=l && r<=qr){
        seg[id].tag += v;
        return;
    }
    push(id);
    int m = (l+r)/2;
    update(ql,qr,v,l,m,id*2+1);
    update(ql,qr,v,m+1,r,id*2+2);
    pull(id);
}

int main(){
    int n;
    cin>>n;
    vector<array<int,4>>Q(2*n);
    
    int L,R,D,U;
    for(int i=0;i<n;i++){
        cin>>L>>R>>D>>U;
        Q[2*i] = {D,L,R-1,1};
        Q[2*i+1] = {U,L,R-1,-1};
    }
    sort(Q.begin(),Q.end());
    Build(0,SZ-1,0);

    long long ans = 0;

    int id = 0,tmp;
    array<long long,2> pre = {0,0},nw;
    while(id<2*n){
        update(Q[id][1],Q[id][2],Q[id][3],0,SZ-1,0);
        tmp = Q[id++][0];
        while(id<2*n && Q[id][0]==tmp){
            update(Q[id][1],Q[id][2],Q[id][3],0,SZ-1,0);
            id++;
        }
        nw = {tmp,SZ - (get(0)==0? seg[0].cnt : 0)};
        ans += pre[1]*(nw[0] - pre[0]);
        pre = nw;
    }
    cout<<ans;
    return(0);
}
```
